rem Architecture/concept: D Das (December 2025, as recorded by the project owner)
rem Implementation history: maintained in this repository's Git commits

rem HOW THIS HYBRID FILE STARTS
rem   1. cmd.exe runs only this small batch-language launcher.
rem   2. The launcher stores this file's absolute path in POWERSEARCH_SELF.
rem   3. PowerShell reads the same file, finds the final #PS-BEGIN marker, and
rem      evaluates everything after it as PowerShell.
rem   4. -NoProfile avoids machine-specific profile code and -STA supplies the
rem      single-threaded apartment required by Windows Forms and the clipboard.

rem   Double-click PowerSearch.cmd, or run it from Command Prompt. No arguments,

rem '#' is not a batch comment marker. cmd.exe reaches the PowerShell call below,
rem while PowerShell deliberately extracts and executes the section after the
rem marker. LastIndexOf prevents an explanatory mention above from being used.
powershell.exe -NoProfile -STA -Command "$s=[IO.File]::ReadAllText($env:POWERSEARCH_SELF); $m=[char]35+'PS-BEGIN'; $i=$s.LastIndexOf($m); if($i -lt 0){throw 'Embedded PowerShell section not found'}; iex $s.Substring($i+$m.Length)"

# Embedded PowerShell application
#
# Purpose
#   Search a selected folder tree by file/folder name, content, extension, size,
#   and Modified/Created date. Results can be filtered, sorted, opened, copied,
#   and exported to CSV. Text and extensionless files are supported, as are the
#   XML text parts of DOCX, XLSX, and PPTX files.
#
# Privacy and persistence
#   Searches are performed locally. The application sends no data anywhere.
#   Only preferences and recent search terms are stored, in
#   %APPDATA%\PowerSearch\settings.json.
#
# Safety/performance boundaries
#   Results stop at 5,000; content matches stop at 20 per file; regex evaluation
#   has a two-second timeout per line; recursion stops after 60 levels; and
#   cloud-only placeholders are not opened because doing so would download them.
# ============================================================================

# Load saved preferences over safe defaults. Missing, obsolete, or malformed
# properties are ignored so an older settings file cannot prevent startup.
function Get-Settings {
    $default = [pscustomobject]@{
        Folder            = [Environment]::GetFolderPath("MyDocuments")
        NameHistory       = @()
        ContentHistory    = @()
        ExtensionText     = "All files"
        IncludeSubfolders = $true
        AdvancedVisible   = $false
        ExcludeFolders    = ".git,node_modules,.venv,venv"
        UseRegex          = $false
        CaseSensitive     = $false
        UseNameWildcards  = $false
        SimilarSpelling   = $true
        MatchMode         = "Exact phrase"
        ShowContext       = $false
        ContextLines      = 2
        TryOtherTextTypes = $true
        DateFilterField   = "Modified"
    }
    if (Test-Path -LiteralPath $script:SettingsPath) {
        try {
            $loaded = Get-Content -LiteralPath $script:SettingsPath -Raw -Encoding UTF8 | ConvertFrom-Json
            foreach ($p in $default.PSObject.Properties.Name) {
                if ($loaded.PSObject.Properties.Name -contains $p -and $null -ne $loaded.$p) { $default.$p = $loaded.$p }
            }
        } catch { }
    }
    return $default
}

# Persist user preferences as local JSON. Failure is intentionally non-fatal:
# searching should still work when the profile directory is read-only.
function Save-Settings {
    param($Settings)
    try {
        if (-not (Test-Path -LiteralPath $script:SettingsDir)) { New-Item -ItemType Directory -Path $script:SettingsDir -Force | Out-Null }
        $Settings | ConvertTo-Json -Depth 5 | Set-Content -LiteralPath $script:SettingsPath -Encoding UTF8
    } catch { }
}

# Put a nonblank search at the front of a de-duplicated, size-limited history.
function Add-HistoryEntry {
    param([string[]]$History, [string]$Entry, [int]$Max = 15)
    if ([string]::IsNullOrWhiteSpace($Entry)) { return @($History) }
    $list = New-Object Collections.Generic.List[string]
    [void]$list.Add($Entry)
    foreach ($h in @($History)) { if ($h -and $h -ne $Entry -and -not $list.Contains($h)) { [void]$list.Add($h) } }
    if ($list.Count -gt $Max) { $list = $list.GetRange(0, $Max) }
    return @($list)
}

# Extract human-readable XML text from Office Open XML ZIP packages. This is a
# lightweight text search, not a full Word/Excel/PowerPoint rendering engine.
function Get-OpenXmlText {
    param([string]$Path)
    $zip = $null
    try {
        $zip = [System.IO.Compression.ZipFile]::OpenRead($Path)
        $wanted = switch ([IO.Path]::GetExtension($Path).ToLowerInvariant()) {
            ".docx" { @("word/document.xml", "word/header", "word/footer") }
            ".xlsx" { @("xl/sharedStrings.xml", "xl/worksheets/") }
            ".pptx" { @("ppt/slides/") }
            default { @() }
        }
        $builder = New-Object Text.StringBuilder
        foreach ($entry in $zip.Entries) {
            $include = $false
            foreach ($prefix in $wanted) {
                if ($entry.FullName -eq $prefix -or $entry.FullName.StartsWith($prefix)) { $include = $true; break }
            }
            if (-not $include -or -not $entry.FullName.EndsWith(".xml")) { continue }
            $reader = New-Object IO.StreamReader($entry.Open())
            try { $xmlText = $reader.ReadToEnd() } finally { $reader.Dispose() }
            $plain = $xmlText -replace '</(w:p|a:p|si|row)>', "`n" -replace '<[^>]+>', ' '
            [void]$builder.AppendLine([Net.WebUtility]::HtmlDecode($plain))
        }
        return $builder.ToString()
    }
    catch {
        return ""
    }
    finally {
        if ($zip) { $zip.Dispose() }
    }
}

# Detect online-only cloud placeholders without opening or hydrating the file.
function Test-IsCloudPlaceholder {
    param([System.IO.FileInfo]$File)
    # OneDrive/Dropbox/Google Drive "Files On-Demand" mark a file that has not been
    # downloaded to this PC with these Cloud Files API attributes. Opening or reading
    # such a file forces Windows to download it first -- for a large file or a slow
    # connection that can take a very long time, and being one single blocking .NET
    # call, Cancel cannot interrupt it once it has started. Checking Attributes is
    # metadata already returned by folder enumeration, so this costs no extra I/O
    # and never triggers a download itself.
    $RecallOnOpen = 0x00040000
    $RecallOnDataAccess = 0x00400000
    $attr = [int]$File.Attributes
    return (($attr -band $RecallOnOpen) -ne 0) -or (($attr -band $RecallOnDataAccess) -ne 0)
}

# Decide whether an unknown/extensionless file is probably text by sampling its
# first bytes. Alternating NUL bytes are allowed because they indicate UTF-16.
function Test-LikelyTextFile {
    param([IO.FileInfo]$File)
    if ($script:TextExtensions -contains $File.Extension.ToLowerInvariant() -or
        $script:OfficeExtensions -contains $File.Extension.ToLowerInvariant()) { return $true }
    # Extensionless files (logs, Dockerfile, README, LICENSE, ...) fall through to the
    # same byte-sniffing below rather than being assumed text -- a genuinely binary
    # file with no extension should not be scanned as if it were readable.
    $stream = $null
    try {
        $stream = New-Object IO.FileStream($File.FullName, [IO.FileMode]::Open, [IO.FileAccess]::Read,
            ([IO.FileShare]::ReadWrite -bor [IO.FileShare]::Delete))
        $buffer = New-Object byte[] 4096
        $count = $stream.Read($buffer, 0, $buffer.Length)
        if ($count -eq 0) { return $true }
        $zeroEven = 0; $zeroOdd = 0
        for ($i = 0; $i -lt $count; $i++) {
            if ($buffer[$i] -eq 0) { if (($i % 2) -eq 0) { $zeroEven++ } else { $zeroOdd++ } }
        }
        # NULs usually mean binary; regular NULs on one byte position indicate UTF-16.
        if (($zeroEven + $zeroOdd) -eq 0) { return $true }
        return (($zeroEven -gt ($count / 8)) -xor ($zeroOdd -gt ($count / 8)))
    }
    catch { return $false }
    finally { if ($stream) { $stream.Dispose() } }
}

# Search one file and return structured line matches. Handles ordinary phrases,
# all/any-word modes, regex, simple one-deletion fuzzy matches, and context lines.
function Find-FileContentMatches {
    param(
        [IO.FileInfo]$File,
        [string]$Pattern,
        [bool]$UseRegex,
        [bool]$CaseSensitive,
        [string]$MatchMode = "Exact phrase",
        [bool]$SimilarSpelling = $true,
        [int]$ContextLines = 0,
        [int]$MaximumMatches = 20
    )

    $comparison = if ($CaseSensitive) { [StringComparison]::Ordinal } else { [StringComparison]::OrdinalIgnoreCase }
    $terms = @($Pattern -split '\s+' | Where-Object { $_ })
    $alternatives = @()
    if ($SimilarSpelling -and -not $UseRegex -and $Pattern.Length -ge 5 -and $Pattern -notmatch '\s') {
        for ($i = 0; $i -lt $Pattern.Length; $i++) { $alternatives += $Pattern.Remove($i, 1) }
    }
    $regex = $null
    if ($UseRegex) {
        $options = if ($CaseSensitive) { [Text.RegularExpressions.RegexOptions]::None } else { [Text.RegularExpressions.RegexOptions]::IgnoreCase }
        # A timeout stops a catastrophic-backtracking pattern (e.g. "(a+)+$") from
        # hanging the search forever on one line; Cancel only gets checked between
        # lines/files, not inside an in-flight IsMatch(), so without this a bad
        # pattern is otherwise unrecoverable short of force-closing the app.
        $regex = New-Object Text.RegularExpressions.Regex($Pattern, $options, [TimeSpan]::FromSeconds(2))
    }

    # Returns whether a line matches, and whether the match only came from a
    # close-spelling alternative (so the caller can flag it as non-exact).
    $lineMatches = {
        param([string]$Line)
        if ($UseRegex) {
            try { return [pscustomobject]@{ Match = $regex.IsMatch($Line); Fuzzy = $false } }
            catch [Text.RegularExpressions.RegexMatchTimeoutException] {
                $script:RegexTimeoutCount++
                return [pscustomobject]@{ Match = $false; Fuzzy = $false }
            }
        }
        if ($MatchMode -eq "All words") {
            foreach ($term in $terms) { if ($Line.IndexOf($term, $comparison) -lt 0) { return [pscustomobject]@{ Match = $false; Fuzzy = $false } } }
            return [pscustomobject]@{ Match = ($terms.Count -gt 0); Fuzzy = $false }
        }
        if ($MatchMode -eq "Any word") {
            foreach ($term in $terms) { if ($Line.IndexOf($term, $comparison) -ge 0) { return [pscustomobject]@{ Match = $true; Fuzzy = $false } } }
            return [pscustomobject]@{ Match = $false; Fuzzy = $false }
        }
        if ($Line.IndexOf($Pattern, $comparison) -ge 0) { return [pscustomobject]@{ Match = $true; Fuzzy = $false } }
        foreach ($alternative in $alternatives) { if ($Line.IndexOf($alternative, $comparison) -ge 0) { return [pscustomobject]@{ Match = $true; Fuzzy = $true } } }
        return [pscustomobject]@{ Match = $false; Fuzzy = $false }
    }

    $reader = $null; $stream = $null
    $resultsFound = New-Object Collections.ArrayList
    $before = New-Object 'Collections.Generic.Queue[string]'
    $pending = New-Object Collections.ArrayList
    try {
        if ($script:OfficeExtensions -contains $File.Extension.ToLowerInvariant()) {
            $reader = New-Object IO.StringReader((Get-OpenXmlText -Path $File.FullName))
        } else {
            # Read files even while another application is appending or rotating them.
            $stream = New-Object IO.FileStream($File.FullName, [IO.FileMode]::Open, [IO.FileAccess]::Read,
                ([IO.FileShare]::ReadWrite -bor [IO.FileShare]::Delete), 65536, [IO.FileOptions]::SequentialScan)
            $reader = New-Object IO.StreamReader($stream, [Text.Encoding]::UTF8, $true, 65536, $true)
        }
        $lineNumber = 0
        while (($line = $reader.ReadLine()) -ne $null -and -not $script:CancelRequested) {
            $lineNumber++
            foreach ($waiting in @($pending)) {
                [void]$waiting.Context.Add($line)
                $waiting.Remaining--
                if ($waiting.Remaining -le 0) {
                    $waiting.Preview = $(if ($waiting.Fuzzy) { "~ " } else { "" }) + ($waiting.Context -join "  |  ")
                    [void]$resultsFound.Add($waiting)
                    [void]$pending.Remove($waiting)
                }
            }
            if (($resultsFound.Count + $pending.Count) -lt $MaximumMatches) {
                $hit = & $lineMatches $line
                if ($hit.Match) {
                    $context = New-Object Collections.ArrayList
                    foreach ($prior in $before) { [void]$context.Add($prior) }
                    [void]$context.Add($line)
                    $match = [pscustomobject]@{ LineNumber=$lineNumber; MatchLine=$line; Context=$context; Remaining=$ContextLines; Preview=""; Fuzzy=$hit.Fuzzy }
                    if ($ContextLines -eq 0) {
                        $match.Preview = $(if ($hit.Fuzzy) { "~ " } else { "" }) + $line
                        [void]$resultsFound.Add($match)
                    }
                    else { [void]$pending.Add($match) }
                }
            }
            $before.Enqueue($line)
            while ($before.Count -gt $ContextLines) { [void]$before.Dequeue() }
            if (($lineNumber % 250) -eq 0) { [Windows.Forms.Application]::DoEvents() }
            if ($resultsFound.Count -ge $MaximumMatches -and $pending.Count -eq 0) { break }
        }
        foreach ($waiting in @($pending)) {
            $waiting.Preview = $(if ($waiting.Fuzzy) { "~ " } else { "" }) + ($waiting.Context -join "  |  ")
            [void]$resultsFound.Add($waiting)
        }
        return @($resultsFound)
    }
    finally {
        if ($reader) { $reader.Dispose() }
        if ($stream) { $stream.Dispose() }
    }
}

# Convert a byte count into a compact, human-readable value for the UI.
function Format-Size {
    param([long]$Bytes)
    if ($Bytes -ge 1GB) { return "{0:N2} GB" -f ($Bytes / 1GB) }
    if ($Bytes -ge 1MB) { return "{0:N2} MB" -f ($Bytes / 1MB) }
    if ($Bytes -ge 1KB) { return "{0:N1} KB" -f ($Bytes / 1KB) }
    return "$Bytes B"
}

# Normalize file/folder metadata and optional content-match data into the single
# row model used by filtering, sorting, display, opening, and CSV export.
function New-ResultRow {
    param(
        [System.IO.FileSystemInfo]$File,
        [Nullable[int]]$LineNumber = $null,
        [string]$Preview = "",
        [string]$MatchLine = ""
    )
    $cleanPreview = ($Preview -replace '\s+', ' ').Trim()
    $displayPreview = $cleanPreview
    if ($displayPreview.Length -gt 300) { $displayPreview = $displayPreview.Substring(0, 297) + "..." }
    $isFolder = $File -is [System.IO.DirectoryInfo]
    # DirectoryInfo has no DirectoryName property (only FileInfo does), so a
    # folder result would otherwise get Folder = $null and later crash
    # Test-RowMatchesFilter's $Row.Folder.IndexOf(...) as soon as the user
    # types into "Filter results" (or exports with a filter active).
    $parentFolder = if ($isFolder) { $(if ($File.Parent) { $File.Parent.FullName } else { "" }) } else { $File.DirectoryName }
    return [pscustomobject]@{
        Name            = $File.Name
        Folder          = $parentFolder
        LineNumber      = $LineNumber
        LineDisplay     = $(if ($null -eq $LineNumber) { "" } else { $LineNumber.ToString("N0") })
        PreviewFull     = $cleanPreview
        PreviewDisplay  = $displayPreview
        MatchLine       = $MatchLine
        SizeBytes       = $(if ($isFolder) { -1 } else { $File.Length })
        SizeDisplay     = $(if ($isFolder) { "" } else { Format-Size $File.Length })
        TypeDisplay     = $(if ($isFolder) { "Folder" } else { $File.Extension.TrimStart('.').ToUpperInvariant() })
        Modified        = $File.LastWriteTime
        ModifiedDisplay = $File.LastWriteTime.ToString("yyyy-MM-dd HH:mm")
        Created         = $File.CreationTime
        CreatedDisplay  = $File.CreationTime.ToString("yyyy-MM-dd HH:mm")
        Path            = $File.FullName
        IsFolder        = $isFolder
    }
}

# Convert the internal row model into the columns displayed by the ListView.
function New-ResultListViewItem {
    param($Row)
    $item = New-Object System.Windows.Forms.ListViewItem($Row.Name)
    [void]$item.SubItems.Add($Row.Folder)
    [void]$item.SubItems.Add($Row.LineDisplay)
    [void]$item.SubItems.Add($Row.PreviewDisplay)
    [void]$item.SubItems.Add($Row.SizeDisplay)
    [void]$item.SubItems.Add($Row.TypeDisplay)
    [void]$item.SubItems.Add($Row.ModifiedDisplay)
    [void]$item.SubItems.Add($Row.CreatedDisplay)
    $item.Tag = $Row
    if ($Row.PreviewFull -and $Row.PreviewFull -ne $Row.PreviewDisplay) { $item.ToolTipText = $Row.PreviewFull }
    return $item
}

# Apply the live "Filter shown results" text to already-found result rows.
function Test-RowMatchesFilter {
    param($Row, [string]$FilterText)
    if ([string]::IsNullOrWhiteSpace($FilterText)) { return $true }
    return ($Row.Name.IndexOf($FilterText, [StringComparison]::OrdinalIgnoreCase) -ge 0) -or
           ($Row.Folder.IndexOf($FilterText, [StringComparison]::OrdinalIgnoreCase) -ge 0) -or
           ($Row.PreviewFull.IndexOf($FilterText, [StringComparison]::OrdinalIgnoreCase) -ge 0)
}

# Rebuild the visible list from the complete result set, applying the current
# display filter and selected sort column without rerunning the disk search.
function Render-Results {
    $filterText = $txtFilterResults.Text.Trim()
    # Windows PowerShell 5.1 has a DLR binder bug where wrapping a
    # List[object] directly in @(...) throws "Argument types do not match".
    # .ToArray() sidesteps the buggy conversion path entirely.
    $rows = $script:AllRows.ToArray()
    if ($script:SortField) {
        $rows = @($rows | Sort-Object -Property $script:SortField -Descending:(-not $script:SortAscending))
    }
    $results.BeginUpdate()
    try {
        $results.Items.Clear()
        $shown = 0
        foreach ($row in $rows) {
            if (Test-RowMatchesFilter -Row $row -FilterText $filterText) {
                [void]$results.Items.Add((New-ResultListViewItem $row))
                $shown++
            }
        }
    } finally { $results.EndUpdate() }
    $lblResultCount.Text = if ($script:AllRows.Count -eq 0) { "" }
        elseif ($shown -eq $script:AllRows.Count) { "$shown result(s)" }
        else { "$shown of $($script:AllRows.Count) result(s)" }
}

# Add one normalized result to the master collection and visible result list.
function Add-ResultRow {
    param(
        [System.IO.FileSystemInfo]$File,
        [Nullable[int]]$LineNumber = $null,
        [string]$Preview = "",
        [string]$MatchLine = ""
    )
    $row = New-ResultRow -File $File -LineNumber $LineNumber -Preview $Preview -MatchLine $MatchLine
    [void]$script:AllRows.Add($row)
    [void]$results.Items.Add((New-ResultListViewItem $row))
}

# Permit a folder dragged from Explorer to become the search root.
$txtFolder.Add_DragEnter({
    param($srcObj, $e)
    if ($e.Data.GetDataPresent([Windows.Forms.DataFormats]::FileDrop)) { $e.Effect = [Windows.Forms.DragDropEffects]::Copy }
    else { $e.Effect = [Windows.Forms.DragDropEffects]::None }
})
$txtFolder.Add_DragDrop({
    param($srcObj, $e)
    $paths = $e.Data.GetData([Windows.Forms.DataFormats]::FileDrop)
    if ($paths -and $paths.Count -gt 0) {
        $p = [string]$paths[0]
        if (Test-Path -LiteralPath $p -PathType Container) { $txtFolder.Text = $p }
        elseif (Test-Path -LiteralPath $p -PathType Leaf) { $txtFolder.Text = [IO.Path]::GetDirectoryName($p) }
    }
})

# Clicking a column selects it for sorting; clicking it again reverses direction.
$results.Add_ColumnClick({
    param($src, $e)
    $fieldMap = @("Name", "Folder", "LineNumber", "PreviewFull", "SizeBytes", "TypeDisplay", "Modified", "Created")
    if ($e.Column -ge $fieldMap.Count) { return }
    $field = $fieldMap[$e.Column]
    if ($script:SortField -eq $field) { $script:SortAscending = -not $script:SortAscending }
    else { $script:SortField = $field; $script:SortAscending = $true }
    Render-Results
})

# Export the currently displayed (already filtered/sorted) rows to a CSV chosen
# by the user. Export does not silently overwrite a predetermined file.
$btnExport.Add_Click({
    if ($script:AllRows.Count -eq 0) {
        Show-Message "There are no results to export yet." "Export results" ([Windows.Forms.MessageBoxIcon]::Information)
        return
    }
    $dlg = New-Object Windows.Forms.SaveFileDialog
    $dlg.Filter = "CSV file (*.csv)|*.csv|All files (*.*)|*.*"
    $dlg.FileName = "PowerSearch-results.csv"
    if ($dlg.ShowDialog() -ne [Windows.Forms.DialogResult]::OK) { return }
    try {
        $filterText = $txtFilterResults.Text.Trim()
        $rows = @($script:AllRows | Where-Object { Test-RowMatchesFilter -Row $_ -FilterText $filterText })
        if ($script:SortField) { $rows = @($rows | Sort-Object -Property $script:SortField -Descending:(-not $script:SortAscending)) }
        $rows | Select-Object @{N='Name';E={$_.Name}}, @{N='Folder';E={$_.Folder}}, @{N='Line';E={$_.LineDisplay}},
            @{N='MatchingText';E={$_.PreviewFull}}, @{N='Size';E={$_.SizeDisplay}}, @{N='Type';E={$_.TypeDisplay}},
            @{N='Modified';E={$_.ModifiedDisplay}}, @{N='Created';E={$_.CreatedDisplay}}, @{N='FullPath';E={$_.Path}} |
            Export-Csv -LiteralPath $dlg.FileName -NoTypeInformation -Encoding UTF8
        $status.Text = "Exported $($rows.Count) result(s) to $($dlg.FileName)."
    }
    catch {
        Show-Message "Could not save the export file.`n`n$($_.Exception.Message)" "Export results" ([Windows.Forms.MessageBoxIcon]::Error)
    }
})

    $ok.Add_Click({
        if ($kind.SelectedIndex -le 3 -and [string]::IsNullOrWhiteSpace($value.Text)) {
            Show-Message "Enter the word or words you want to find." "Create a search pattern" ([Windows.Forms.MessageBoxIcon]::Warning)
            $value.Focus(); return
        }
        $escaped = [Text.RegularExpressions.Regex]::Escape($value.Text.Trim())
        $made = switch ($kind.SelectedIndex) {
            0 { "\b$escaped\b" }
            1 { "\b$escaped\w*" }
            2 { "\b\w*$escaped\b" }
            3 {
                $words = $value.Text.Split(',') | ForEach-Object { [regex]::Escape($_.Trim()) } | Where-Object { $_ }

        $builder.Close()
    })

    $builder.AcceptButton = $ok
    $builder.Add_Shown({ $value.Focus() })
    [void]$builder.ShowDialog($form)
    $builder.Dispose()
})

    if ($results.SelectedItems.Count -gt 0) {
        $path = [string]$results.SelectedItems[0].Tag.Path

            catch { Show-Message "Windows could not open this item.`n`n$($_.Exception.Message)" "Unable to open" ([Windows.Forms.MessageBoxIcon]::Warning) }
        }
    }
}

# ---- Search pipeline ---------------------------------------------------------
# Validate input, save the current options, enumerate folders iteratively, apply
# inexpensive metadata filters before content reads, collect matches, and keep
# the UI responsive with periodic event processing and cancellation checks.
$btnSearch.Add_Click({
    $root = $txtFolder.Text.Trim()
    if (-not (Test-Path -LiteralPath $root -PathType Container)) {
        Show-Message -Text "Please choose a valid folder." -Icon ([System.Windows.Forms.MessageBoxIcon]::Warning)
        return
    }

    # Remember this search immediately so it is available even if the user closes without another search.
    if ($txtName.Text -and ($txtName.Items -notcontains $txtName.Text)) { [void]$txtName.Items.Insert(0, $txtName.Text) }
    if ($txtContent.Text -and ($txtContent.Items -notcontains $txtContent.Text)) { [void]$txtContent.Items.Insert(0, $txtContent.Text) }

    $namePattern = $txtName.Text.Trim()
    $useCreatedDate = $cmbDateField.Text -eq "Created"

    $extensions = @()
    $typeText = $txtExt.Text.Trim()
    $typeValues = switch ($typeText) {
        "All files"       { "" }
        "Documents"       { "doc,docx,pdf,rtf,odt" }
        "Spreadsheets"    { "xls,xlsx,csv,tsv,ods" }
        "Presentations"   { "ppt,pptx,odp" }
        "Pictures"        { "jpg,jpeg,png,gif,bmp,tif,tiff,webp,heic,svg" }
        "Audio and video" { "mp3,wav,m4a,flac,mp4,mov,avi,mkv,wmv" }
        "Web pages"       { "html,htm,url,css,js" }
        "Text and notes"  { "txt,md,log,ini,rtf" }
        "Archives"        { "zip,7z,rar,gz,tar" }
        default            { $typeText }
    }
    if (-not [string]::IsNullOrWhiteSpace($typeValues)) {
        $extensions = $typeValues.Split(@(',', ';', ' '), [StringSplitOptions]::RemoveEmptyEntries) |
            ForEach-Object { $_.Trim().TrimStart(".").ToLowerInvariant() } |
            Where-Object { $_ }
    }

    $excluded = @()
    if (-not [string]::IsNullOrWhiteSpace($txtExclude.Text)) {
        $excluded = $txtExclude.Text.Split(",") |
            ForEach-Object { $_.Trim().ToLowerInvariant() } |
            Where-Object { $_ }
    }

    if ($chkRegex.Checked -and -not [string]::IsNullOrWhiteSpace($txtContent.Text)) {
        try { [void][Text.RegularExpressions.Regex]::IsMatch("", $txtContent.Text) }
        catch {
            Show-Message -Text "The advanced text pattern is not valid. Please correct it and try again." -Icon ([Windows.Forms.MessageBoxIcon]::Warning)
            $txtContent.Focus(); return
        }
    }

    $minBytes = $null
    $maxBytes = $null

    if (-not [string]::IsNullOrWhiteSpace($txtMinSize.Text)) {
        $n = 0.0
        if (-not [double]::TryParse($txtMinSize.Text, [ref]$n)) {
            Show-Message -Text "Min MB must be a number." -Icon ([System.Windows.Forms.MessageBoxIcon]::Warning)
            return
        }
        $minBytes = [long]($n * 1MB)
    }

    if (-not [string]::IsNullOrWhiteSpace($txtMaxSize.Text)) {
        $n = 0.0
        if (-not [double]::TryParse($txtMaxSize.Text, [ref]$n)) {
            Show-Message -Text "Max MB must be a number." -Icon ([System.Windows.Forms.MessageBoxIcon]::Warning)
            return
        }
        $maxBytes = [long]($n * 1MB)
    }

    $results.Items.Clear()
    $script:AllRows = New-Object Collections.Generic.List[object]
    $script:SortField = $null
    $script:SortAscending = $true
    $txtFilterResults.Text = ""
    $lblResultCount.Text = ""
    $script:CancelRequested = $false
    $btnSearch.Enabled = $false
    $btnCancel.Enabled = $true
    $form.UseWaitCursor = $true
    $progress.Visible = $true
    $sw = [Diagnostics.Stopwatch]::StartNew()

    $matched = 0
    $scanned = 0
    $skippedErrors = 0
    $contentUnsupported = 0
    $filesWithMatches = 0
    $truncatedFiles = 0
    $limitReached = $false
    $cloudSkipped = 0
    $script:RegexTimeoutCount = 0

    try {
        # Safe folder traversal compatible with Windows PowerShell 5.1 / .NET Framework.
        # It skips inaccessible folders instead of aborting the whole search. Loop
        # protection uses a visited-folder set rather than excluding reparse points,
        # because cloud-sync placeholders (OneDrive, Dropbox, Google Drive) are also
        # marked as reparse points and are ordinary folders that must still be searched.
        # The visited set only catches an exact repeated path, not a symlink/junction
        # cycle (each pass through the link produces a longer, never-before-seen path),
        # so a hard depth backstop below is what actually stops those loops.
        $maxFolderDepth = 60
        $depthLimitSkipped = 0
        $visitedFolders = New-Object 'Collections.Generic.HashSet[string]' ([StringComparer]::OrdinalIgnoreCase)
        $folders = New-Object System.Collections.Stack
        $rootDir = New-Object System.IO.DirectoryInfo($root)
        [void]$visitedFolders.Add($rootDir.FullName)
        $folders.Push([pscustomobject]@{ Dir = $rootDir; Depth = 0 })

        while ($folders.Count -gt 0 -and -not $script:CancelRequested -and -not $limitReached) {
            $currentEntry = $folders.Pop()
            $currentDir = [System.IO.DirectoryInfo]$currentEntry.Dir
            $currentDepth = $currentEntry.Depth

            # Folder names are useful results too. A folder is a candidate whenever the
            # active filters are all things a folder can actually have: a name, and
            # optionally a modified-date range. File type, size, and content filters
            # rule folders out because they do not apply to folders.
            $folderCandidate = [string]::IsNullOrWhiteSpace($txtContent.Text) -and $extensions.Count -eq 0 -and
                $null -eq $minBytes -and $null -eq $maxBytes
            if ($folderCandidate -and -not [string]::IsNullOrWhiteSpace($namePattern) -and $currentDir.FullName -ne $root) {
                $folderMatches = if ($chkNamePattern.Checked) { $currentDir.Name -like $namePattern } else { $currentDir.Name.IndexOf($namePattern, [StringComparison]::OrdinalIgnoreCase) -ge 0 }
                $folderFilterDate = if ($useCreatedDate) { $currentDir.CreationTime } else { $currentDir.LastWriteTime }
                if ($folderMatches -and
                    (-not $chkFrom.Checked -or $folderFilterDate.Date -ge $dateFrom.Value.Date) -and
                    (-not $chkTo.Checked -or $folderFilterDate.Date -le $dateTo.Value.Date)) {
                    Add-ResultRow $currentDir; $matched++
                    if ($matched -ge 5000) { $limitReached = $true; continue }
                }
            }

            try {
                $dirFiles = $currentDir.GetFiles()
            }
            catch {
                $skippedErrors++
                continue
            }

            foreach ($fi in $dirFiles) {
                if ($script:CancelRequested) { break }

                $scanned++
                if (($scanned % 75) -eq 0) {
                    $status.Text = "Scanned $scanned files - $matched matches"
                    $lblResultCount.Text = "$($script:AllRows.Count) result(s)"
                    [System.Windows.Forms.Application]::DoEvents()
                }

                try {

                # Excluded folder names
                if ($excluded.Count -gt 0) {
                    $relative = $fi.DirectoryName.Substring($root.TrimEnd('\').Length).TrimStart('\')
                    if ($relative) {
                        $parts = $relative.Split('\')
                        $skip = $false
                        foreach ($part in $parts) {
                            if ($excluded -contains $part.ToLowerInvariant()) {
                                $skip = $true
                                break
                            }
                        }
                        if ($skip) { continue }
                    }
                }

                # Ordinary searches match any part of the name. Wildcards are opt-in.
                if (-not [string]::IsNullOrWhiteSpace($namePattern)) {
                    $nameMatches = if ($chkNamePattern.Checked) { $fi.Name -like $namePattern } else { $fi.Name.IndexOf($namePattern, [StringComparison]::OrdinalIgnoreCase) -ge 0 }
                    if (-not $nameMatches) { continue }
                }

                # Extension filter
                if ($extensions.Count -gt 0) {
                    $ext = $fi.Extension.TrimStart(".").ToLowerInvariant()
                    if ($extensions -notcontains $ext) { continue }
                }

                # Size
                if ($null -ne $minBytes -and $fi.Length -lt $minBytes) { continue }
                if ($null -ne $maxBytes -and $fi.Length -gt $maxBytes) { continue }

                # Modified/Created date range (per the "using" dropdown)
                if ($chkFrom.Checked -or $chkTo.Checked) {
                    $fileFilterDate = if ($useCreatedDate) { $fi.CreationTime } else { $fi.LastWriteTime }
                    if ($chkFrom.Checked -and $fileFilterDate.Date -lt $dateFrom.Value.Date) { continue }
                    if ($chkTo.Checked -and $fileFilterDate.Date -gt $dateTo.Value.Date) { continue }
                }

                # Content
                if (-not [string]::IsNullOrWhiteSpace($txtContent.Text)) {
                    # Never open a file OneDrive/Dropbox/Google Drive hasn't downloaded
                    # yet -- doing so forces a download that Cancel cannot interrupt.
                    # See Test-IsCloudPlaceholder for why this check is safe/free.
                    if (Test-IsCloudPlaceholder $fi) { $cloudSkipped++; continue }
                    $contentExt = $fi.Extension.ToLowerInvariant()
                    if ([string]::IsNullOrEmpty($contentExt)) {
                        # Extensionless files (logs, Dockerfile, README, ...) are searched
                        # by default, but must still pass the same byte-sniff used for
                        # unknown extensions -- a genuine binary with no extension should
                        # not be scanned as if it were text.
                        if (-not (Test-LikelyTextFile $fi)) { $contentUnsupported++; continue }
                    }
                    else {
                        $knownText = ($script:TextExtensions + $script:OfficeExtensions) -contains $contentExt
                        if (-not $knownText -and (-not $chkOtherText.Checked -or -not (Test-LikelyTextFile $fi))) {
                            $contentUnsupported++
                            continue
                        }
                    }
                    $contextCount = if ($chkContext.Checked) { [int]$numContext.Value } else { 0 }
                    $contentMatches = @(Find-FileContentMatches -File $fi -Pattern $txtContent.Text `
                        -UseRegex $chkRegex.Checked -CaseSensitive $chkCase.Checked `
                        -MatchMode $cmbMatchMode.Text -SimilarSpelling $chkSimilar.Checked `
                        -ContextLines $contextCount -MaximumMatches 20)
                    if ($contentMatches.Count -eq 0) { continue }
                    $filesWithMatches++
                    foreach ($contentMatch in $contentMatches) {
                        Add-ResultRow -File $fi -LineNumber $contentMatch.LineNumber `
                            -Preview $contentMatch.Preview -MatchLine $contentMatch.MatchLine
                        $matched++
                        if ($matched -ge 5000) { $limitReached = $true; break }
                    }
                    if ($contentMatches.Count -ge 20) { $truncatedFiles++ }
                    if ($limitReached) { break }
                    continue
                }

                Add-ResultRow -File $fi
                $matched++

                if ($matched -ge 5000) {
                    $status.Text = "Stopped at 5,000 results. Narrow the search for more precise results."
                    $limitReached = $true
                    break
                }
                }
                catch {
                    $skippedErrors++
                }
            }

            if ($chkSubfolders.Checked -and -not $script:CancelRequested) {
                try {
                    foreach ($subdir in $currentDir.GetDirectories()) {
                        if ($excluded.Count -gt 0 -and ($excluded -contains $subdir.Name.ToLowerInvariant())) {
                            continue
                        }

                        # An exact repeated path (e.g. a hardlinked directory reached two
                        # ways) is skipped here. A symlink/junction cycle is not caught by
                        # this check -- each pass through the link produces a new, longer
                        # path -- so the depth cap below is the real backstop for that case.
                        if (-not $visitedFolders.Add($subdir.FullName)) { continue }

                        if ($currentDepth -ge $maxFolderDepth) { $depthLimitSkipped++; continue }

                        $folders.Push([pscustomobject]@{ Dir = $subdir; Depth = $currentDepth + 1 })
                    }
                }
                catch {
                    $skippedErrors++
                }
            }
        }

        $elapsed = " ($([Math]::Round($sw.Elapsed.TotalSeconds, 1))s)"
        if ($script:CancelRequested) {
            $status.Text = "Cancelled. $matched matches from $scanned files scanned.$elapsed"
        }
        elseif ($limitReached) {
            $status.Text += $elapsed
        }
        else {
            $status.Text = "Done. $matched matches from $scanned files scanned."
            if ($filesWithMatches -gt 0) { $status.Text = "Done. $matched matching lines in $filesWithMatches files ($scanned files checked)." }
            if ($matched -eq 0) {
                if ($scanned -eq 0) {
                    $status.Text = "No files were found in '$root'. Choose the folder that contains your file."
                } else {
                    $status.Text = "No matches in $scanned files under '$root'. Check the folder or try fewer words."
                }
            }
            if ($skippedErrors -gt 0) {
                $status.Text += " $skippedErrors inaccessible/problem files skipped."
            }
            if ($contentUnsupported -gt 0) {
                $status.Text += " $contentUnsupported files of unsupported types were not opened."
            }
            if ($cloudSkipped -gt 0) {
                $status.Text += " $cloudSkipped file(s) not downloaded to this PC were skipped instead of downloading them."
            }
            if ($truncatedFiles -gt 0) {
                $status.Text += " Showing the first 20 matches in $truncatedFiles heavily-matching file(s)."
            }
            if ($depthLimitSkipped -gt 0) {
                $status.Text += " $depthLimitSkipped folder(s) skipped past a $maxFolderDepth-level depth limit (possible symlink loop)."
            }
            if ($script:RegexTimeoutCount -gt 0) {
                $status.Text += " $($script:RegexTimeoutCount) line(s) skipped because the pattern took too long against them."
            }
            $status.Text += $elapsed
        }
        $lblResultCount.Text = if ($script:AllRows.Count -eq 0) { "" } else { "$($script:AllRows.Count) result(s)" }
    }
    catch {
        Show-Message -Text $_.Exception.Message -Title "Search error" -Icon ([System.Windows.Forms.MessageBoxIcon]::Error)
        $status.Text = "Search failed."
    }
    finally {
        $btnSearch.Enabled = $true
        $btnCancel.Enabled = $false
        $form.UseWaitCursor = $false
        $progress.Visible = $false
        # The window was closed while this search was running (see
        # Add_FormClosing): the search has now stopped cleanly, so replay the
        # close. $btnCancel.Enabled is $false at this point, so FormClosing
        # will let it through instead of deferring again.
        if ($script:ClosePending) { $form.Close() }
    }
})

# Save the final UI state and recent terms when the window closes. If a search
# is still running, defer the close instead: request cancellation the same
# cooperative way the Cancel button does, and let the search's own `finally`
# block replay Close() once the loop has actually stopped. Without this, the
# window would disappear while the search kept running against disposed
# controls in the background -- which looks like "Close hangs".
$form.Add_FormClosing({
    if ($btnCancel.Enabled) {
        # Still running (also covers a repeat close click while a previous one
        # is already pending): keep deferring rather than letting this attempt
        # fall through to a real close, which would strand the search again.
        $_.Cancel = $true
        $script:ClosePending = $true
        $script:CancelRequested = $true
        $status.Text = "Closing..."
        return
    }

    $s = $script:Settings
    $s.Folder = $txtFolder.Text
    $s.NameHistory = Add-HistoryEntry -History $txtName.Items -Entry $txtName.Text
    $s.ContentHistory = Add-HistoryEntry -History $txtContent.Items -Entry $txtContent.Text
    $s.ExtensionText = $txtExt.Text
    $s.IncludeSubfolders = $chkSubfolders.Checked
    $s.AdvancedVisible = $grpFilterFiles.Visible
    $s.ExcludeFolders = $txtExclude.Text
    $s.UseRegex = $chkRegex.Checked
    $s.CaseSensitive = $chkCase.Checked
    $s.UseNameWildcards = $chkNamePattern.Checked
    $s.SimilarSpelling = $chkSimilar.Checked
    $s.MatchMode = $cmbMatchMode.Text
    $s.ShowContext = $chkContext.Checked
    $s.ContextLines = [int]$numContext.Value
    $s.TryOtherTextTypes = $chkOtherText.Checked
    $s.DateFilterField = $cmbDateField.Text
    Save-Settings -Settings $s
})

$dateFieldIdx = $cmbDateField.Items.IndexOf($script:Settings.DateFilterField)

    $numContext.Value = $clamped

# Live CPU/RAM readout for this process, refreshed once a second. CPU% is
# computed the same way Task Manager does: processor time consumed since the
# last sample, divided by wall-clock time elapsed and by core count.
$script:UsageProcess = [Diagnostics.Process]::GetCurrentProcess()
$script:LastCpuSampleTime = Get-Date
$script:LastCpuTime = $script:UsageProcess.TotalProcessorTime
$resourceTimer = New-Object System.Windows.Forms.Timer
$resourceTimer.Interval = 1000
$resourceTimer.Add_Tick({
    try {
        $script:UsageProcess.Refresh()
        $now = Get-Date
        $cpuNow = $script:UsageProcess.TotalProcessorTime
        $elapsedMs = ($now - $script:LastCpuSampleTime).TotalMilliseconds
        $cpuPercent = 0
        if ($elapsedMs -gt 0) {
            $cpuPercent = [Math]::Max(0, [Math]::Round((($cpuNow - $script:LastCpuTime).TotalMilliseconds / $elapsedMs / [Environment]::ProcessorCount) * 100))
        }
        $script:LastCpuTime = $cpuNow
        $script:LastCpuSampleTime = $now
        $lblResourceUsage.Text = "CPU {0}%  RAM {1}" -f $cpuPercent, (Format-Size $script:UsageProcess.WorkingSet64)
    } catch { }
})
$resourceTimer.Start()
$form.Add_FormClosed({ $resourceTimer.Stop() })

Set-AdvancedPanelVisible -Show ([bool]$script:Settings.AdvancedVisible)
